title: "1684. Count the Number of Consistent Strings"
tags: [leetcode]

```cpp
class Solution {
public:
    int countConsistentStrings(string allowed, vector<string>& words)
    {
        unordered_set<int> st(allowed.begin(), allowed.end());
        int cnt = words.size();
        for (auto word : words)
        {
            for (auto w : word)
            {
                if (!st.count(w))
                {
                    cnt--;
                    break;
                }
            }
        }
        return cnt;
    }
};
```
- T: $O(M + N \cdot K)$
- S: $O(M)$
