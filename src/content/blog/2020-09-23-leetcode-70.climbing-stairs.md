```cpp
class Solution {
public:
    int climbStairs(int n)
    {
        if(n < 2) return n;

        vector<int> dp(n + 1);

        dp[0] = 1, dp[1] = 1;

        for(int i = 2; i <= n; i++)
        {
            dp[i] = dp[i - 1] + dp[i - 2];
        }
        return dp[n];
    }
};
```

- T: $O(N)$
- S: $O(N)$

## DP improved

```cpp
class Solution {
public:
    int climbStairs(int n)
    {
        int previousStep = 1, currentStep = 1;
        for(int step = 2; step <= n; step++)
        {
            int nextStep = previousStep + currentStep;
            previousStep = currentStep;
            currentStep = nextStep;
        }
        return currentStep;
    }
};
```

- T: $O(N)$
- S: $O(1)$
