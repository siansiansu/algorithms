```cpp
class Solution {
public:
    int countBalls(int lowLimit, int highLimit) {
        unordered_map<int, int> m;
        int res = 0;
        for (int i = lowLimit; i <= highLimit; i++) {
            m[digitsSum(i)]++;
        }
        for (const auto& [k, v] : m) {
            res = max(res, v);
        }
        return res;
    }
    int digitsSum(int n) {
        int sum = 0;
        while (n) {
            sum += n % 10;
            n /= 10;
        }
        return sum;
    }
};
```
