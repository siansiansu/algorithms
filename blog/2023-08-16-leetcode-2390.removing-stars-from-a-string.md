```cpp
class Solution {
public:
    string removeStars(string s) {
        int cnt = 0;
        string res = "";
        for (int i = s.size() - 1; i >= 0; i--) {
            if (s[i] == '*') {
                cnt++;
            } else if (cnt != 0 && s[i] != '*') {
                cnt--; continue;
            } else {
                res += s[i];
            }
        }
        reverse(res.begin(), res.end());
        return res;
    }
};
```
