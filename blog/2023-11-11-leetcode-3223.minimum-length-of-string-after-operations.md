title: "3223. Minimum Length of String After Operations"
tags: [leetcode]
---

```cpp
class Solution {
public:
    int minimumLength(string s)
    {
        int n = s.size();
        vector<int> freq(26);
        for (char c : s)
        {
            ++freq[c - 'a'];
            if (freq[c - 'a'] == 3)
            {
                freq[c - 'a'] -= 2;
                n -= 2;
            }
        }
        return n;
    }
};
```

- T: $O(N)$
- S: $O(1)$
